def tool(name,args):
 req=urllib.request.Request('http://127.0.0.1:9472/mcp',data=json.dumps({'jsonrpc':'2.0','id':1,'method':'tools/call','params':{'name':name,'arguments':args}}).encode(),headers={'Content-Type':'application/json'})
 data=json.load(urllib.request.urlopen(req,timeout=20))
 if data.get('error'):raise RuntimeError(data['error'])
 result=data['result']
 if result.get('isError'):raise RuntimeError(result['content'])
 try:return json.loads(result['content'][0]['text'])
 except json.JSONDecodeError:return result['content'][0]['text']
