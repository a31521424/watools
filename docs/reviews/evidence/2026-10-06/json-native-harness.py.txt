ids={}
for path in (root/'frontend/bindings/watools/internal/host').glob('*service.ts'):
    ids.update({name:int(number) for name,number in re.findall(r'export function (\w+)\([^\n]*\{\n\s+return \$Call.ByID\((\d+)',path.read_text())})
def tool(name,args):
    req=urllib.request.Request(url,data=json.dumps({'jsonrpc':'2.0','id':1,'method':'tools/call','params':{'name':name,'arguments':args}}).encode(),headers={'Content-Type':'application/json'})
    response=json.load(urllib.request.urlopen(req,timeout=20))
    if response.get('error'): raise RuntimeError(response['error'])
    result=response['result']
    if result.get('isError'): raise RuntimeError(result['content'])
    return json.loads(result['content'][0]['text'])
def js(code):
    result=tool('js_eval',{'window':'launcher','js':'try {'+code+'} catch (error) {return {qaError: error.message}}','timeout_ms':12000})
    if isinstance(result,dict) and result.get('qaError'):raise RuntimeError(result['qaError'])
    return result
def wait(fn, timeout=20):
    deadline=time.monotonic()+timeout
    while time.monotonic()<deadline:
        try:
            result=fn()
            if result:return result
        except Exception:
            if host.poll() is not None:raise RuntimeError('native process exited')
        time.sleep(.1)
    raise RuntimeError('native readiness condition timed out')
