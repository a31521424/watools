func qaCString(value string, fn func(*C.char) int) int {
	c := C.CString(value)
	defer C.free(unsafe.Pointer(c))
	return fn(c)
}
func qaHandle() unsafe.Pointer {
	desktop := application.Get()
	if desktop == nil {
		return nil
	}
	window, ok := desktop.Window.GetByName("launcher")
	if !ok {
		return nil
	}
	return window.(*application.WebviewWindow).NativeWindow()
}
func init() {
	go func() {
		mux := http.NewServeMux()
		mux.HandleFunc("/", func(w http.ResponseWriter, r *http.Request) {
			w.Header().Set("Content-Type", "application/json")
			var value any
			err := fmt.Errorf("unknown probe route")
			switch r.URL.Path {
			case "/state":
				value = application.InvokeSyncWithResult(func() json.RawMessage {
					c := C.qaWindowInfo(qaHandle())
					defer C.free(unsafe.Pointer(c))
					return json.RawMessage(C.GoString(c))
				})
				err = nil
			case "/hotkey/interleave":
 err=app.AcceptanceLifecycleInterleave();value=1
 case "/hotkey/fire":
 err=app.AcceptanceFireRegisteredHotkey();value=1
 case "/hotkey":
				if r.Method == "POST" {
					var cfg app.HotkeyConfigAPI
					if decode := json.NewDecoder(http.MaxBytesReader(w, r.Body, 4096)).Decode(&cfg); decode != nil {
						err = decode
					} else {
						err = app.GetWaApp().UpdateHotkey(cfg)
					}
				} else {
					err = nil
				}
				value = map[string]any{"configs": app.GetWaApp().GetHotkeys(), "registered": app.GetHotkeyManager().IsRegistered("show-hide-window")}
			case "/menu":
				value = application.InvokeSyncWithResult(func() int {
					return qaCString(r.URL.Query().Get("title"), func(c *C.char) int { return int(C.qaMainMenuAction(c)) })
				})
				err = nil
			case "/clipboard/backup":
				value = application.InvokeSyncWithResult(func() int { return int(C.qaBackupClipboard()) })
				err = nil
			case "/clipboard/restore":
				value = application.InvokeSyncWithResult(func() int { return int(C.qaRestoreClipboard()) })
				err = nil
			case "/clipboard/fixture":
				kind := r.URL.Query().Get("kind")
				fixture := filepath.Join(os.Getenv("WATOOLS_QA_FIXTURES"), "clipboard.txt")
				if kind == "image" {
					fixture = filepath.Join(os.Getenv("WATOOLS_QA_FIXTURES"), "clipboard.png")
				}
				value = application.InvokeSyncWithResult(func() int {
					return qaCString(kind, func(k *C.char) int {
						return qaCString(fixture, func(p *C.char) int { return int(C.qaClipboardFixture(k, p)) })
					})
				})
				err = nil
			case "/dialog":
				value = application.InvokeSyncWithResult(func() json.RawMessage {
					c := C.qaDialogInfo(qaHandle())
					defer C.free(unsafe.Pointer(c))
					return json.RawMessage(C.GoString(c))
				})
				err = nil
			case "/dialog/cancel":
				value = application.InvokeSyncWithResult(func() int { return int(C.qaCancelDialog(qaHandle())) })
				err = nil
			case "/snapshot":
				value = application.InvokeSyncWithResult(func() int {
					return qaCString(filepath.Join(os.Getenv("WATOOLS_QA_FIXTURES"), "launcher.png"), func(c *C.char) int { return int(C.qaSnapshot(qaHandle(), c)) })
				})
				err = nil
			case "/snapshot/state":
				value = application.InvokeSyncWithResult(func() int { return int(C.qaSnapshotState()) })
				err = nil
			case "/activate":
 value = application.InvokeSyncWithResult(func() int {return int(C.qaActivateLegacy(qaHandle()))});err=nil
 case "/focus":
				var target struct{ PID int }
				if decode := json.NewDecoder(http.MaxBytesReader(w, r.Body, 4096)).Decode(&target); decode != nil {
					err = decode
				} else {
					value = application.InvokeSyncWithResult(func() int { return int(C.qaFocusProcess(C.int(target.PID))) })
					err = nil
				}
			}
			if err != nil {
				w.WriteHeader(400)
				json.NewEncoder(w).Encode(map[string]any{"error": err.Error(), "state": value})
				return
			}
			json.NewEncoder(w).Encode(value)
		})
		http.ListenAndServe("127.0.0.1:9248", mux)
	}()
}
